cmake_minimum_required(VERSION 3.20)
project(Web2Exe VERSION 1.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(WEB2EXE_BUILD_CLI "Compila o CLI (Linux/Termux)" ON)
option(WEB2EXE_BUILD_APP "Compila o app Windows (cross)" OFF)
option(WEB2EXE_WINE "Modo compatível com Wine (msvcrt + fallback sem WebView2)" OFF)
set(WEB2EXE_BUILD_INSTALLER OFF CACHE BOOL "Instalador é gerado por scripts/make-installer.sh")

if(WEB2EXE_BUILD_CLI)
  add_executable(web2exe src/cli/main.cpp)
  target_include_directories(web2exe PRIVATE src/cli)
  target_compile_definitions(web2exe PRIVATE WEB2EXE_VERSION="${PROJECT_VERSION}")
  if(UNIX)
    target_compile_options(web2exe PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()

if(WEB2EXE_BUILD_APP)
  enable_language(RC)
  set(WEB2EXE_GENERATED_APP_DIR "${CMAKE_SOURCE_DIR}/src/app")
  if(DEFINED WEB2EXE_GENERATED_APP_DIR_OVERRIDE)
    set(WEB2EXE_GENERATED_APP_DIR "${WEB2EXE_GENERATED_APP_DIR_OVERRIDE}")
  endif()
  add_executable(web2exe_app WIN32
    "${WEB2EXE_GENERATED_APP_DIR}/main.cpp"
    "${WEB2EXE_GENERATED_APP_DIR}/resources.rc")
  target_include_directories(web2exe_app PRIVATE "${CMAKE_SOURCE_DIR}/src/app/compat" "${WEBVIEW2_SDK}/build/native/include")
  target_compile_definitions(web2exe_app PRIVATE UNICODE _UNICODE WEB2EXE_VERSION="${PROJECT_VERSION}")
  if(WEB2EXE_WINE)
    target_compile_definitions(web2exe_app PRIVATE WEB2EXE_WINE=1)
  endif()
  target_link_libraries(web2exe_app PRIVATE user32 ole32 oleaut32 shell32 shlwapi advapi32 version uuid)
  target_link_options(web2exe_app PRIVATE -municode -s)
endif()
